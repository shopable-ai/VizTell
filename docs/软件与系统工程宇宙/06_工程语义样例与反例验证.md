## 3. 已执行的最小区分性实验

B-N 为负例，B-P 为对应正例。各项显式计算实际输出并与预置 expected 比较。负例“通过”指成功识别反例，不是接受非法输入。N01 双可编辑事实；N02 独立源集合漏成员但内部组合全覆盖；N03 新版本/hash；N04 同条件同优先级相反规则；N05 非法端点/未知关系；N06 外部结果未知；N07 声明超过关键路径等级或 pending；N08 scope 扩张。所有数据均合成；A案例真实文件只静态核对，没有运行实际实现。N02 中 site:other 是人为注入的源成员，不是对历史项目真实遗漏的断言。N04 只识别相同条件，不处理逻辑蕴含或任意规则求解。N07 ordinal 4/5/7 只在本实验中表示 fixture_runtime_proven/local_runtime_proven/production_proof（零基序），不是新增枚举。

复验：把以下 JavaScript 原样保存到临时文件并在 Node.js 执行（例如 node fixture.js）；不依赖数据库、网络、包。原始函数在本轮 Code Mode JS 直接执行，不是从日志倒填结果。此研究 fixture 只检验必要性质，未运行 L1 检查器、未证明正式关系注册或复杂传播算法正确。

```javascript
function runResearchFixture() {
  const results=[];
  const check=(id,expected,actual)=>{results.push({id,expected,actual,pass:JSON.stringify(expected)===JSON.stringify(actual)})};
  const duplicates=rows=>rows.filter((r,i)=>rows.slice(0,i).some(p=>p.key===r.key&&p.path!==r.path&&p.editable&&r.editable)).length;
  const fact={key:"project:rule@1",path:"requirements#K1.4",editable:true};
  check("B-N01",1,duplicates([fact,{...fact,path:"chain#behavior",editable:true}]));
  check("B-P01",0,duplicates([fact,{...fact,path:"chain#behavior",editable:false}]));
  const fullMembers=["duration:valid","duration:unknown","site:local","site:iyf","site:other"];
  const combinations=[["valid","local"],["valid","iyf"],["unknown","local"],["unknown","iyf"]];
  const coverage=(domainA,domainB,rows)=>{const pairs=domainA.flatMap(a=>domainB.map(b=>JSON.stringify([a,b])));const hits=new Set(rows.map(x=>JSON.stringify(x)));return {covered:pairs.filter(x=>hits.has(x)).length,total:pairs.length}};
  const inventory=["duration:valid","duration:unknown","site:local","site:iyf"];
  const missing=(source,derived)=>source.filter(x=>!derived.includes(x));
  check("B-N02",{covered:4,total:4,missing:["site:other"]},{...coverage(["valid","unknown"],["local","iyf"],combinations),missing:missing(fullMembers,inventory)});
  check("B-P02",{covered:6,total:6,missing:[]},{...coverage(["valid","unknown"],["local","iyf","other"],[...combinations,["valid","other"],["unknown","other"]]),missing:missing(fullMembers,fullMembers)});
  const evidence={subject:"rule",version:1,hash:"fixture-h1",env:"fixture",verified:true,fresh:true,route:"fixture-route",scope:["interval"]};
  const applies=(e,s)=>e.verified&&e.fresh&&["subject","version","hash","env","route"].every(k=>e[k]===s[k])&&s.scope.every(x=>e.scope.includes(x));
  check("B-N03",false,applies(evidence,{...evidence,version:2,hash:"fixture-h2"}));
  check("B-P03",true,applies(evidence,{...evidence}));
  const conflicting=(rules)=>rules.some((a,i)=>rules.slice(i+1).some(b=>a.subject===b.subject&&a.condition===b.condition&&a.priority===b.priority&&a.action!==b.action));
  const skip={subject:"outro",condition:"same-time-same-episode",priority:1,action:"skip"};
  check("B-N04",true,conflicting([skip,{...skip,action:"keep"}]));
  check("B-P04",false,conflicting([skip,{...skip,condition:"different-episode",action:"keep"}]));
  const endpointPairs={"derives_from":[["fact","source"],["requirement","fact"],["requirement","requirement"]],"implements":[["task","spec"],["function","business_capability"]]};
  const legal=(type,a,b)=>!!endpointPairs[type]?.some(([s,t])=>a===s&&b===t);
  check("B-N05",[false,false], [legal("implements","task","source"),legal("invented_relation","task","spec")]);
  check("B-P05",true,legal("implements","task","spec"));
  const nextAction=op=>op.outcome==="unknown"&&!op.independentReconciled?"reconcile_blocked":op.outcome==="not_applied"&&op.reconciled&&op.freshAuthorization?"retry_new_attempt":"no_retry";
  check("B-N06","reconcile_blocked",nextAction({outcome:"unknown",independentReconciled:false,freshAuthorization:true}));
  check("B-P06","retry_new_attempt",nextAction({outcome:"not_applied",reconciled:true,freshAuthorization:true}));
  // Local ordinal projection of two existing proof levels, not a new proof enum.
  const canClaim=(verdict,critical,requested)=>verdict==="accepted"&&critical.length>0&&requested<=Math.min(...critical);
  check("B-N07",[false,false], [canClaim("accepted",[4,5],7),canClaim("pending",[7],4)]);
  check("B-P07",true,canClaim("accepted",[4,5],4));
  check("B-N08",false,applies(evidence,{...evidence,scope:["interval","real_site"]}));
  check("B-P08",true,applies(evidence,{...evidence,scope:["interval"]}));
  return results;
}
const result = runResearchFixture();
console.log(JSON.stringify(result, null, 2));
if (result.some(x => !x.pass)) throw new Error("research fixture failed");
```
